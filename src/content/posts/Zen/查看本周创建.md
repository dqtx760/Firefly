```dataview
TABLE WITHOUT ID
  category AS "分类",
  dateformat(file.ctime, "yyyy/MM/dd") AS "创建日期",
  file.link AS "本周笔记"
WHERE file.ctime >= date(today) - dur(3 days)
SORT category ASC, file.ctime DESC
```
