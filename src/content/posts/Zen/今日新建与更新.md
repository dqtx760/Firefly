```dataview
TABLE WITHOUT ID
  file.link AS "今日创建",
  dateformat(file.ctime, "yyyy/MM/dd HH:mm") AS "创建时间"
WHERE file.cday = date(today)
SORT file.ctime desc
```

### 今日更新
```dataview
TABLE WITHOUT ID
  file.link AS "今日更新",
  dateformat(file.mtime, "yyyy/MM/dd HH:mm") AS "更新时间"
WHERE file.mday = date(today)
SORT file.mtime desc
```
