```dataview
TABLE WITHOUT ID
  dateformat(file.ctime, "yyyy/MM/dd") AS "日期",
  file.link AS "笔记",
  category AS "主题",
  join(file.tags, " ") AS "tags"
WHERE 
  contains(file.folder, "posts")
  AND file.ctime >= date(today) - dur(7 days)
SORT file.ctime DESC, category ASC
```
